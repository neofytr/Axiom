Training Loop Guide

A step-by-step walkthrough of writing a complete training loop in Axiom, from data loading through evaluation.

Overview

Training a neural network consists of repeating these steps:

  1. Load a batch of data
  2. Run the forward pass to get predictions
  3. Compute the loss
  4. Run the backward pass to compute gradients
  5. Update the weights
  6. Clean up the computation graph

The sections below build up a complete example piece by piece.

Step 1: Initialize and Prepare Data

#include "axiom/axiom.h"
#include <stdio.h>

int main(void) {
    ax_init();
    ax_set_seed(42);

    // load your data. here we assume train_x is [N, 784] and
    // train_y is [N, 10] (one-hot encoded labels)
    ax_tensor_t *train_x = ...;  // your input data
    ax_tensor_t *train_y = ...;  // your labels

    // normalize inputs to zero mean, unit variance
    ax_tensor_t *norm_x = ax_transform_normalize(train_x);

    // wrap as a dataset and create a dataloader
    ax_dataset_t *ds = ax_tensor_dataset_create(norm_x, train_y);
    ax_dataloader_t *dl = ax_dataloader_create(ds, 256, true);

Step 2: Build the Network

    // 784 -> 256 -> 128 -> 10 with batchnorm and dropout
    ax_layer_t *net = ax_sequential_create();

    ax_sequential_add(net, ax_dense_create(784, 256, true));
    ax_sequential_add(net, ax_batchnorm_create(256, 1e-5f, 0.1f));
    ax_sequential_add(net, ax_relu_layer_create());
    ax_sequential_add(net, ax_dropout_create(0.3f));

    ax_sequential_add(net, ax_dense_create(256, 128, true));
    ax_sequential_add(net, ax_batchnorm_create(128, 1e-5f, 0.1f));
    ax_sequential_add(net, ax_relu_layer_create());
    ax_sequential_add(net, ax_dropout_create(0.2f));

    ax_sequential_add(net, ax_dense_create(128, 10, true));

    ax_layer_summary(net);

Step 3: Create Model, Optimizer, and Scheduler

    ax_model_t *model = ax_model_create(net);

    ax_optimizer_t *opt = ax_adam_create(
        model->params, model->n_params,
        1e-3f,    // lr
        0.9f,    // beta1
        0.999f,  // beta2
        1e-8f,   // eps
        1e-4f); // weight decay

    ax_model_compile(model, opt, ax_cross_entropy_loss);

    // cosine LR schedule: decay from 1e-3 to 1e-5 over 50 epochs
    int n_epochs = 50;
    ax_lr_scheduler_t *sched = ax_sched_cosine(opt, n_epochs, 1e-5f);

Step 4: The Training Loop

    for (int epoch = 0; epoch < n_epochs; epoch++) {
        ax_layer_train(net);        // enable dropout, batchnorm training mode
        ax_dataloader_reset(dl);    // reshuffle the data

        float epoch_loss = 0;
        int n_batches = 0;
        ax_batch_t batch;

        while (ax_dataloader_next(dl, &batch)) {
            // single train step: forward -> loss -> backward -> step -> cleanup
            float loss = ax_model_train_step(model, batch.input, batch.target);
            epoch_loss += loss;
            n_batches++;

            // dataloader allocates batch tensors; we must free them
            ax_tensor_destroy(batch.input);
            ax_tensor_destroy(batch.target);
        }

        // step the LR scheduler
        ax_sched_step(sched);

        printf("epoch %3d  loss %.4f  lr %.6f\n",
               epoch, epoch_loss / n_batches, ax_sched_get_lr(sched));
    }

Step 5: Evaluation

    // switch to eval mode: dropout disabled, batchnorm uses running stats
    ax_layer_eval(net);
    ax_no_grad();  // skip gradient tracking for speed

    ax_tensor_t *pred = ax_model_predict(model, test_x);
    // ... compute accuracy, print results ...

    ax_enable_grad();
    ax_tensor_destroy(pred);

Step 6: Save and Clean Up

    // save the trained model
    ax_model_save(model, "trained.axm");

    // clean up
    ax_sched_destroy(sched);
    ax_dataloader_destroy(dl);
    ax_dataset_destroy(ds);
    ax_tensor_destroy(norm_x);
    ax_tensor_destroy(train_x);
    ax_tensor_destroy(train_y);
    ax_model_destroy(model);  // frees net, opt, and all params
    ax_shutdown();
    return 0;
}

The Manual Training Loop

If you need more control than ax_model_train_step() provides (e.g., for gradient clipping or custom loss combinations), here is the full manual loop:

while (ax_dataloader_next(dl, &batch)) {
    // 1. zero gradients
    ax_optimizer_zero_grad(opt);

    // 2. forward
    ax_tensor_t *pred = ax_layer_forward(net, batch.input);

    // 3. loss
    ax_tensor_t *loss = ax_cross_entropy_loss(pred, batch.target);
    float loss_val = ax_tensor_get_f32(loss, (int64_t[]){0});

    // 4. backward
    ax_backward(loss);

    // 5. gradient clipping (optional)
    ax_clip_grad_norm(model->params, model->n_params, 1.0f);

    // 6. update weights
    ax_optimizer_step(opt);

    // 7. clean up computation graph (frees all intermediate tensors)
    ax_graph_cleanup(loss);
    ax_tensor_destroy(loss);

    // 8. clean up batch
    ax_tensor_destroy(batch.input);
    ax_tensor_destroy(batch.target);
}

Tips

  • Always call ax_graph_cleanup() after backward. Without it, every intermediate tensor from the forward pass leaks.
  • Destroy batch tensors. ax_dataloader_next() allocates new tensors for each batch. You own them and must destroy them.
  • BatchNorm needs train/eval mode. In training mode it computes batch statistics and updates running stats. In eval mode it uses the running stats. Forgetting to switch causes accuracy drops at test time.
  • Gradient clipping helps deep networks. Use ax_clip_grad_norm() with a max norm of 1.0 for stability.
  • LR scheduling matters. Cosine annealing or warmup+cosine almost always beats a constant learning rate.
  • Weight decay in AdamW is better than L2 regularization in Adam. Use ax_adamw_create() if you want regularization.

Loading from CSV

// load a CSV with 4 feature columns and 1 target column
ax_dataset_t *ds = ax_csv_dataset_load(
    "data.csv",
    (int[]){0, 1, 2, 3}, 4,  // feature columns
    (int[]){4}, 1,              // target column
    true);                      // has header row

ax_dataloader_t *dl = ax_dataloader_create(ds, 32, true);
